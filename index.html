// URL'den veya localStorage'dan kimlik bilgilerini oku const urlParams = new URLSearchParams(window.location.search); let ownerId = urlParams.get('id') || localStorage.getItem('ice_owner_id'); let storedPin = localStorage.getItem('ice_pin_' + ownerId) || ''; async function fetchStudioData() { if (!ownerId) { showLoginModal("Please provide your Avatar UUID to access Studio."); return; } const endpoint = `${BACKEND_URL}/api/settings?id=${ownerId}&pin=${encodeURIComponent(storedPin)}`; try { const response = await fetch(endpoint); if (response.status === 401) { // PIN Gerekli - Paneli gizle, PIN şifre kutusunu aç showPinModal(); return; } const data = await response.json(); if (data.status === "success") { // Oturum başarılı - PIN'i yerel belleğe sabitle localStorage.setItem('ice_owner_id', ownerId); if (storedPin) localStorage.setItem('ice_pin_' + ownerId, storedPin); renderPanelUI(data); } } catch (err) { console.error("Connection error:", err); } } // PIN Onaylama Butonu async function handlePinSubmit(inputPin) { const res = await fetch(`${BACKEND_URL}/api/verify-pin`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ ownerId: ownerId, pin: inputPin }) }); const resData = await res.json(); if (res.ok && resData.status === "success") { storedPin = inputPin; localStorage.setItem('ice_pin_' + ownerId, inputPin); hidePinModal(); fetchStudioData(); } else { alert("Incorrect Security PIN. Access Denied."); } } // Yeni PIN Belirleme (Panel içinden) async function handleSetNewPin(newPin) { const res = await fetch(`${BACKEND_URL}/api/set-pin`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ ownerId: ownerId, pin: newPin }) }); const resData = await res.json(); if (res.ok) { storedPin = newPin; localStorage.setItem('ice_pin_' + ownerId, newPin); alert("Remote Access PIN successfully set!"); } else { alert(resData.error || "Failed to set PIN."); } }
